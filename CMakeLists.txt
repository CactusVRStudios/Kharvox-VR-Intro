cmake_minimum_required(VERSION 3.24)
project(VRCracktro LANGUAGES C CXX RC)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Windows x64 is required")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(KHARVOX_OPENXR_LOADER_LIB "${CMAKE_CURRENT_SOURCE_DIR}/build/pc-loader-static/sdk/src/loader/Release/openxr_loader.lib" CACHE FILEPATH "Static OpenXR loader (build with build-release.ps1)")
add_library(KharvoxOpenXR STATIC IMPORTED)
set_target_properties(KharvoxOpenXR PROPERTIES IMPORTED_LOCATION "${KHARVOX_OPENXR_LOADER_LIB}")
target_link_libraries(KharvoxOpenXR INTERFACE advapi32)
add_executable(PcAssetPacker EXCLUDE_FROM_ALL tools/pack_pc_asset.cpp)
target_compile_features(PcAssetPacker PRIVATE cxx_std_17)
target_link_libraries(PcAssetPacker PRIVATE cabinet)
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/pc-assets")
set(packed_assets)
foreach(asset IN ITEMS 04B_30__.TTF ninja.mod Copper_41.png)
  set(output "${CMAKE_CURRENT_BINARY_DIR}/pc-assets/${asset}.bin")
  add_custom_command(OUTPUT "${output}"
    COMMAND PcAssetPacker "${CMAKE_CURRENT_SOURCE_DIR}/assets/cracktro/vr/${asset}" "${output}"
    DEPENDS PcAssetPacker "${CMAKE_CURRENT_SOURCE_DIR}/assets/cracktro/vr/${asset}" VERBATIM)
  list(APPEND packed_assets "${output}")
endforeach()
add_custom_target(PcPackedAssets DEPENDS ${packed_assets})
option(KHARVOX_COMPRESS_ASSETS "Compress resources separately (disable when packing the entire EXE)" ON)
if(KHARVOX_COMPRESS_ASSETS)
  set(INTRO_FONT_ASSET "${CMAKE_CURRENT_BINARY_DIR}/pc-assets/04B_30__.TTF.bin")
  set(INTRO_MUSIC_ASSET "${CMAKE_CURRENT_BINARY_DIR}/pc-assets/ninja.mod.bin")
  set(INTRO_COPPER_ASSET "${CMAKE_CURRENT_BINARY_DIR}/pc-assets/Copper_41.png.bin")
else()
  set(INTRO_FONT_ASSET "${CMAKE_CURRENT_SOURCE_DIR}/assets/cracktro/vr/04B_30__.TTF")
  set(INTRO_MUSIC_ASSET "${CMAKE_CURRENT_SOURCE_DIR}/assets/cracktro/vr/ninja.mod")
  set(INTRO_COPPER_ASSET "${CMAKE_CURRENT_SOURCE_DIR}/assets/cracktro/vr/Copper_41.png")
endif()
configure_file(src/intro/VrGameIntro.rc.in "${CMAKE_CURRENT_BINARY_DIR}/VrGameIntro.rc" @ONLY)
set(intro_music src/intro/IntroMusic.cpp src/intro/Pocketmod.c)
set(intro_resource "${CMAKE_CURRENT_BINARY_DIR}/VrGameIntro.rc")
function(configure_intro target)
  if(KHARVOX_COMPRESS_ASSETS)
    add_dependencies(${target} PcPackedAssets)
    target_compile_definitions(${target} PRIVATE KHARVOX_COMPRESSED_ASSETS)
  endif()
  target_compile_features(${target} PRIVATE cxx_std_17)
  if(MSVC)
    target_compile_options(${target} PRIVATE $<$<CONFIG:Release>:/O1> $<$<CONFIG:Release>:/Gy> $<$<CONFIG:Release>:/Gw>)
    set_property(TARGET ${target} PROPERTY INTERPROCEDURAL_OPTIMIZATION_RELEASE TRUE)
    target_link_options(${target} PRIVATE $<$<CONFIG:Release>:/OPT:REF> $<$<CONFIG:Release>:/OPT:ICF>)
  endif()
  target_include_directories(${target} PRIVATE third-party/openxr-sdk/include)
  target_link_libraries(${target} PRIVATE d3d11 d3dcompiler windowscodecs dxgi user32 gdi32 xaudio2 ole32 cabinet)
endfunction()
add_executable(VRCracktro WIN32 src/intro/VrGameIntro.cpp ${intro_music} ${intro_resource})
configure_intro(VRCracktro)
target_link_libraries(VRCracktro PRIVATE KharvoxOpenXR)
set_target_properties(VRCracktro PROPERTIES OUTPUT_NAME "KHARVOX Intro")
install(TARGETS VRCracktro RUNTIME DESTINATION .)
include(CTest)
add_executable(QuestAssetBaker EXCLUDE_FROM_ALL tools/bake_quest_assets.cpp ${intro_resource})
configure_intro(QuestAssetBaker)
if(BUILD_TESTING)
  add_executable(IntroLoaderTests tests/openxr_loader_tests.cpp)
  target_compile_features(IntroLoaderTests PRIVATE cxx_std_17)
  target_include_directories(IntroLoaderTests PRIVATE third-party/openxr-sdk/include)
  target_link_libraries(IntroLoaderTests PRIVATE KharvoxOpenXR)
  add_test(NAME intro-loader COMMAND IntroLoaderTests)
  add_executable(IntroRenderTests tests/vr_cube_intro_tests.cpp ${intro_resource})
  configure_intro(IntroRenderTests)
  add_test(NAME intro-render COMMAND IntroRenderTests "${CMAKE_CURRENT_BINARY_DIR}/test-output")
  add_executable(IntroMusicTests tests/intro_music_tests.cpp ${intro_music} ${intro_resource})
  configure_intro(IntroMusicTests)
  add_test(NAME intro-music COMMAND IntroMusicTests "${CMAKE_CURRENT_BINARY_DIR}/test-output")
  add_executable(IntroShutdownTests tests/intro_shutdown_tests.cpp ${intro_music} ${intro_resource})
  configure_intro(IntroShutdownTests)
  target_link_libraries(IntroShutdownTests PRIVATE KharvoxOpenXR)
  add_test(NAME intro-shutdown COMMAND IntroShutdownTests)
